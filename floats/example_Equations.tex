\chapter{Equations}

\section{Single Equations}
\label{sec:equations-single}

\begin{equation}
	\label{eq:equations-basic}
	E = mc^2
\end{equation}

Use \texttt{split} inside \texttt{equation} for one numbered equation that spans
multiple lines.

\begin{equation}
	\label{eq:equations-multiline}
	\begin{split}
		f(x) &= ax^2 + bx + c \\
		     &= a(x-r_1)(x-r_2)
	\end{split}
\end{equation}

Use \verb|\tag| when the printed equation identifier needs to be set manually.

\begin{equation*}
	\nabla \cdot \vect{u} = 0
	\tag{continuity}
	\label{eq:equations-continuity}
\end{equation*}

\section{Matrices and Arrays}

	\begin{array}{rcl}
		\text{left} & \longrightarrow & \text{right}
	\end{array}
	\tag{array}
	\label{eq:equations-array}
\end{equation*}

\begin{equation}
	\label{eq:equations-matrix-system}
	\begin{bmatrix*}[r]
		6.7861 & 2.2313 \\
		2.2313 & 2.8148
	\end{bmatrix*}
	\vect{\ddot{u}}
	+
	\begin{bmatrix*}[r]
		 0.9057 & -0.9057 \\
		-0.9057 &  0.9057
	\end{bmatrix*}
	\vect{u}
	=
	\begin{pmatrix*}[r]
		1 \\
		10
	\end{pmatrix*}
\end{equation}
